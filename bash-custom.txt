# Use a UTF-8 locale when the session has none
if [ "$(locale charmap 2>/dev/null)" != "UTF-8" ]; then
    utf8_locale=$(locale -a 2>/dev/null | grep -ixE 'c\.utf-?8' | head -n 1)
    if [ -z "$utf8_locale" ]; then
        utf8_locale=$(locale -a 2>/dev/null | grep -ixE 'en_us\.utf-?8' | head -n 1)
    fi
    if [ -n "$utf8_locale" ]; then
        export LC_CTYPE="$utf8_locale"
    fi
    unset utf8_locale
fi

if [ -n "${ZSH_VERSION:-}" ]; then
    CURRENT_SHELL="zsh"
    RC_FILE="$HOME/.zshrc"
    eval 'CUSTOM_DIR="${${(%):-%x}:A:h}"'
else
    CURRENT_SHELL="bash"
    RC_FILE="$HOME/.bashrc"
    CUSTOM_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" 2>/dev/null && pwd)"
fi

# ~*x_X- SHRIMP ROTATION -X_x*~
set_shrimps() {
    if [ "$USE_EMOJI" = true ]; then
        SHRIMPS=("🦐" "🍤")
    else
        SHRIMPS=("[>]" "[<]")
    fi
}
set_shrimps
SHRIMP_INDEX=0

rotate_shrimp() {
    # zsh arrays start at 1, bash arrays start at 0
    if [ "$CURRENT_SHELL" = "zsh" ]; then
        CURRENT_SHRIMP="${SHRIMPS[$(( SHRIMP_INDEX + 1 ))]}"
    else
        CURRENT_SHRIMP="${SHRIMPS[$SHRIMP_INDEX]}"
    fi
    SHRIMP_INDEX=$(( (SHRIMP_INDEX + 1) % ${#SHRIMPS[@]} ))
}

ga() {
    git add -A "$@" && git status --short
}

save_setting() {
    local begin='# >>> helpful-bash-additions >>>'
    local end='# <<< helpful-bash-additions <<<'
    local tmp result
    grep -qxF "$begin" "$RC_FILE" 2>/dev/null || return 1
    tmp="$(mktemp "${TMPDIR:-/tmp}/bash-additions.XXXXXX")" || return 1
    awk -v begin="$begin" -v end="$end" -v name="$1" -v value="$2" '
        $0 == begin { inside = 1 }
        $0 == end { inside = 0 }
        inside && index($0, name "=") == 1 { next }
        inside && index($0, "source ") == 1 { print name "=" value }
        { print }
    ' "$RC_FILE" > "$tmp" && { cat "$tmp" > "$RC_FILE"; } 2>/dev/null
    result=$?
    rm -f "$tmp"
    return $result
}

# Pop a specific stash
gpopn() {
    if [ -z "${1:-}" ]; then
        echo "Usage: gpopn <stash-number> or gpn <stash-number>"
        echo "Example: gpopn 0"
        echo "Run glist or gsl to see all stashes"
        return 1
    fi
    git stash pop "stash@{$1}"
}
alias gpn='gpopn'

toggle_emoji() {
    if [ "$USE_EMOJI" = true ]; then
        USE_EMOJI=false
    else
        USE_EMOJI=true
    fi
    set_shrimps
    build_prompt

    local state="enabled"
    [ "$USE_EMOJI" = true ] || state="disabled"
    if save_setting USE_EMOJI "$USE_EMOJI"; then
        echo "Emoji $state"
    else
        echo "Emoji $state for this session only (not saved in $RC_FILE)"
    fi
}

bupdate() {
    if ! git -C "$CUSTOM_DIR" rev-parse --git-dir > /dev/null 2>&1; then
        echo "Can not check for updates: $CUSTOM_DIR is not a git clone"
        return 1
    fi
    echo "Checking for updates..."
    if ! git -C "$CUSTOM_DIR" fetch --quiet 2> /dev/null; then
        echo "Could not check for updates (git fetch failed). Are you online?"
        return 1
    fi
    local behind answer
    behind=$(git -C "$CUSTOM_DIR" rev-list --count 'HEAD..@{upstream}' 2> /dev/null)
    if [ -z "$behind" ]; then
        echo "Can not check for updates: this clone's branch does not track a remote branch"
        return 1
    fi
    if [ "$behind" -eq 0 ]; then
        echo "Already up to date"
        return 0
    fi
    printf "Update available. Download? (y/n) "
    read -r answer
    case "$answer" in
        [Yy]|[Yy][Ee][Ss])
            if git -C "$CUSTOM_DIR" pull --ff-only; then
                echo "Updated. Run bsource or open a new terminal to use it."
            else
                echo "Update failed. Nothing was changed. See git's message above."
                return 1
            fi
            ;;
        *)
            echo "Not updated"
            ;;
    esac
}

bashhelp() {
    local gs_name=" ${SKIP_SHORTCUTS:-} "
    case "$gs_name" in
        *" gs="*)
            gs_name="${gs_name#* gs=}"
            gs_name="${gs_name%% *}"
            ;;
        *)
            gs_name="${GS_ALIAS:-gs}"
            ;;
    esac
    echo "---------------------------------------------------"
    echo "  CUSTOM SHELL ALIASES ($CURRENT_SHELL)"
    echo "---------------------------------------------------"
    echo ""
    echo "GIT:"
    printf "  %-22s - Git status\n" "$gs_name"
    echo "  ga                     - Git add all, then list what is staged"
    echo "  gd/gdiff               - Git diff"
    echo "  gl/glog                - Git log"
    echo "  gb                     - Git branch list"
    echo "  gco <branch>           - Git checkout branch (use -b for new)"
    echo "  gcm 'message'          - Git commit with message"
    echo "  gpo                    - Git push to current branch"
    echo "  gpr/gpull              - Git pull from current branch"
    echo "  gundo                  - Undo last commit (keep changes)"
    echo "  gstash                 - Stash changes"
    echo "  gpop                   - Pop latest stash"
    echo "  gpn/gpopn <n>          - Pop specific stash number"
    echo "  gsl/glist              - List all stashes"
    echo ""
    echo "SHELL:"
    echo "  bsource                - Reload shell config ($RC_FILE)"
    echo "  bopen                  - Edit shell config"
    echo "  dumprc / drc           - Show bash file"
    echo "  toggle_emoji           - Toggle emoji on/off in prompt"
    echo "  bupdate                - Check for updates and download them"
    echo "  bsetup                 - Run setup (bsetup -u to uninstall)"
    echo "  bhelp / bash_help      - Show this help message"
    echo ""
    echo "---------------------------------------------------"
}

# Ask once before using a name another program already has
other_program() {
    if [ "$CURRENT_SHELL" = "zsh" ]; then
        whence -p "$1" 2> /dev/null
    else
        type -P "$1" 2> /dev/null
    fi
}

move_shortcut() {
    local line
    if line="$(alias "$1" 2> /dev/null)"; then
        eval "alias $2=${line#*=}"
        unalias "$1"
    elif line="$(declare -f "$1")"; then
        eval "$2${line#"$1"}"
        unset -f "$1"
    else
        return 1
    fi
}

ask_alias() {
    while true; do
        printf "Alias for %s (Enter for none): " "$1"
        read -r
        case "$REPLY" in
            "")
                return 1
                ;;
            *[!A-Za-z0-9_]*)
                echo "Use only letters, numbers and _"
                ;;
            *)
                if ! type "$REPLY" > /dev/null 2>&1; then
                    return 0
                fi
                echo "$REPLY is already a command"
                ;;
        esac
    done
}

shortcut_question() {
    local name="$1" program text answer setting moved
    shift
    program="$(other_program "$name")"
    if text="$(alias "real_$name" 2> /dev/null)"; then
        text="${text#*=}"
        text="${text#\'}"
        text="${text%\'}"
    fi
    echo "Another program is also installed as $name ($program)"
    printf "Keep %s as %s? (y/n) " "$name" "${text:-a shortcut}"
    read -r answer
    case "$answer" in
        [Yy]|[Yy][Ee][Ss])
            KEEP_SHORTCUTS="${KEEP_SHORTCUTS:+$KEEP_SHORTCUTS }$name"
            setting="KEEP_SHORTCUTS=\"$KEEP_SHORTCUTS\""
            ;;
        [Nn]|[Nn][Oo])
            if ask_alias "${text:-the shortcut}"; then
                moved="$REPLY"
            fi
            SKIP_SHORTCUTS="${SKIP_SHORTCUTS:+$SKIP_SHORTCUTS }$name${moved:+=$moved}"
            setting="SKIP_SHORTCUTS=\"$SKIP_SHORTCUTS\""
            ;;
        *)
            echo "Not changed"
            return 1
            ;;
    esac
    if ! save_setting "${setting%%=*}" "${setting#*=}"; then
        echo "Kept for this session only (not saved in $RC_FILE)"
    fi
    unset -f "$name"
    case "$setting" in
        KEEP*)
            move_shortcut "real_$name" "$name"
            if ask_alias "$program"; then
                alias "$REPLY=command $name"
                if ! save_setting "alias $REPLY" "'command $name'"; then
                    echo "Kept for this session only (not saved in $RC_FILE)"
                fi
                echo "$REPLY now runs $program"
            else
                echo "The other $name is still there: command $name"
            fi
            eval "$name \"\$@\""
            ;;
        *)
            if [ -n "${moved:-}" ]; then
                move_shortcut "real_$name" "$moved"
                echo "$moved is now ${text:-the shortcut}, $name runs $program"
            else
                unalias "real_$name" 2> /dev/null || unset -f "real_$name"
                echo "$name now runs $program"
            fi
            ;;
    esac
}

ask_first() {
    local name moved
    for name in "$@"; do
        moved=" ${SKIP_SHORTCUTS:-} "
        case "$moved" in
            *" $name="*)
                moved="${moved#* $name=}"
                move_shortcut "$name" "${moved%% *}"
                unset -f "$name" 2> /dev/null
                continue
                ;;
        esac
        other_program "$name" > /dev/null || continue
        [ "$name" = "${GS_ALIAS:-}" ] && continue
        case " ${KEEP_SHORTCUTS:-} " in
            *" $name "*) continue ;;
        esac
        case " ${SKIP_SHORTCUTS:-} " in
            *" $name "*)
                unalias "$name" 2> /dev/null
                unset -f "$name" 2> /dev/null
                continue
                ;;
        esac
        move_shortcut "$name" "real_$name" || continue
        eval "$name() { shortcut_question $name \"\$@\"; }"
    done
}
ask_first gs gst ga gd gdiff gl glog gb gco gcm gpo gpr gpull gundo gstash gpop gpn gpopn gsl glist \
    bsource bopen dumprc drc toggle_emoji bupdate bsetup bhelp bash_help bashhelp

safe_git() {
    local overrides=() key
    while IFS= read -r key; do
        overrides+=(-c "$key=")
    done < <(git config --name-only --get-regexp '^filter\..*\.(clean|process)$' 2>/dev/null)
    git --no-optional-locks -c core.fsmonitor=false -c core.hooksPath=/dev/null \
        -c log.showSignature=false "${overrides[@]}" "$@"
}

git_state() {
    PROMPT_BRANCH=""      # Branch name, or "detached"
    PROMPT_HASH=""        # Short commit hash, empty before the first commit
    PROMPT_AHEAD=0
    PROMPT_BEHIND=0
    PROMPT_STASH=0
    PROMPT_CONFLICT=0
    PROMPT_STATE=""       # Staged, dirty or clean, empty outside a repo

    local line staged=0 dirty=0 in_repo=0
    while IFS= read -r line; do
        in_repo=1
        case "$line" in
            "# branch.oid (initial)")
                ;;
            "# branch.oid "*)
                PROMPT_HASH="${line#\# branch.oid }"
                PROMPT_HASH="${PROMPT_HASH:0:7}"
                ;;
            "# branch.head "*)
                PROMPT_BRANCH="${line#\# branch.head }"
                ;;
            "# branch.ab "*)
                # Rendered as # branch.ab +2 -1
                line="${line#\# branch.ab +}"
                PROMPT_AHEAD="${line%% *}"
                PROMPT_BEHIND="${line##*-}"
                ;;
            "# stash "*)
                PROMPT_STASH="${line#\# stash }"
                ;;
            "u "*)
                PROMPT_CONFLICT=1
                ;;
            "? "*)
                dirty=1
                ;;
            [12]" "*)
                # Rendered as 1 M. ...  (first letter staged, second unstaged)
                if [ "${line:2:1}" != "." ]; then staged=1; fi
                if [ "${line:3:1}" != "." ]; then dirty=1; fi
                ;;
        esac
    done < <(safe_git status --porcelain=v2 --branch --show-stash \
                 --ignore-submodules=dirty 2>/dev/null)

    if [ "$in_repo" = 0 ]; then
        return
    fi
    if [ "$PROMPT_BRANCH" = "(detached)" ]; then
        PROMPT_BRANCH="detached"
    fi
    if [ "$staged" = 1 ]; then
        PROMPT_STATE="staged"
    elif [ "$dirty" = 1 ]; then
        PROMPT_STATE="dirty"
    else
        PROMPT_STATE="clean"
    fi
}

if [ "$CURRENT_SHELL" = "zsh" ]; then
    autoload -Uz add-zsh-hook
    if [[ "$(whence -f precmd 2>/dev/null)" == *rotate_shrimp* ]]; then
        unfunction precmd
    fi
    typeset -ga psvar

    update_prompt() {
        rotate_shrimp
        git_state
        local arrows=""
        if [ "$PROMPT_AHEAD" -gt 0 ]; then arrows+=" ↑$PROMPT_AHEAD"; fi
        if [ "$PROMPT_BEHIND" -gt 0 ]; then arrows+=" ↓$PROMPT_BEHIND"; fi
        psvar=("$CURRENT_SHRIMP" "" "" "" "$arrows" "" "" "" "")
        if [ -n "$PROMPT_BRANCH" ]; then psvar[2]="($PROMPT_BRANCH)"; fi
        if [ -n "$PROMPT_HASH" ]; then psvar[3]=" $PROMPT_HASH"; fi
        if [ "$PROMPT_CONFLICT" = 1 ]; then psvar[4]="1"; fi
        case "$PROMPT_STATE" in
            staged) psvar[6]="1" ;;
            dirty)  psvar[7]="1" ;;
            clean)  psvar[8]="1" ;;
        esac
        if [ "$PROMPT_STASH" -gt 0 ]; then psvar[9]="stash:$PROMPT_STASH"; fi
    }
    add-zsh-hook precmd update_prompt

    build_prompt() {
        PROMPT='%1v '                                 # Shrimp
        PROMPT+='%F{240}[%D{%H:%M:%S}]%f '            # Timestamp [HH:MM:SS] (gray)
        PROMPT+='%F{green}%B%n@%m%b%f:'               # user@host (green bold)
        PROMPT+='%F{blue}%B%~%b%f'                    # Current directory (blue bold)
        PROMPT+='%(6V.%F{yellow}%B.)'                 # Staged: yellow bold
        PROMPT+='%(7V.%F{red}.)'                      # Unstaged/untracked: red
        PROMPT+='%(8V.%F{green}.)'                    # Clean: green
        PROMPT+=' %2v%3v'                             # (branch) hash
        PROMPT+='%(4V. %F{magenta}%BCONFLICT%b%f.)'   # Merge conflict
        PROMPT+='%5v'                                 # Ahead/behind arrows
        PROMPT+='%(9V. %F{cyan}%9v%f.)'               # stash:N
        PROMPT+='%f%b%(!.#.$) '                       # Reset color + prompt
    }
    build_prompt
else
    update_prompt() {
        # Start on a new line if the last output had no newline
        printf '%*s\r\033[K' "${COLUMNS:-80}" ''
        rotate_shrimp
        git_state
        local s=$'\001' e=$'\002' esc=$'\033'
        local reset="${s}${esc}[00m${e}"
        PROMPT_GIT=""
        case "$PROMPT_STATE" in
            staged) PROMPT_GIT+="${s}${esc}[1;33m${e}" ;;   # Yellow (staged changes)
            dirty)  PROMPT_GIT+="${s}${esc}[0;31m${e}" ;;   # Red (unstaged/untracked)
            clean)  PROMPT_GIT+="${s}${esc}[0;32m${e}" ;;   # Green (clean)
        esac
        PROMPT_GIT+=" "
        if [ -n "$PROMPT_BRANCH" ]; then PROMPT_GIT+="($PROMPT_BRANCH)"; fi
        if [ -n "$PROMPT_HASH" ]; then PROMPT_GIT+=" $PROMPT_HASH"; fi
        if [ "$PROMPT_CONFLICT" = 1 ]; then
            PROMPT_GIT+=" ${s}${esc}[1;35m${e}CONFLICT${reset}"
        fi
        if [ "$PROMPT_AHEAD" -gt 0 ]; then PROMPT_GIT+=" ↑$PROMPT_AHEAD"; fi
        if [ "$PROMPT_BEHIND" -gt 0 ]; then PROMPT_GIT+=" ↓$PROMPT_BEHIND"; fi
        if [ "$PROMPT_STASH" -gt 0 ]; then
            PROMPT_GIT+=" ${s}${esc}[0;36m${e}stash:$PROMPT_STASH${reset}"
        fi
        PROMPT_GIT+="$reset"
        # bash loses the cursor when the prompt covers more than two rows
        PROMPT_BREAK=""
        if (( BASH_VERSINFO[0] > 4 || (BASH_VERSINFO[0] == 4 && BASH_VERSINFO[1] >= 4) )); then
            local text="${PS1@P}"
            while [[ "$text" == *$'\001'*$'\002'* ]]; do
                text="${text%%$'\001'*}${text#*$'\002'}"
            done
            local ascii="${text//[^[:ascii:]]/}"
            if [ $(( 2 * ${#text} - ${#ascii} )) -ge $(( 2 * ${COLUMNS:-80} )) ]; then
                PROMPT_BREAK=$'\n'
            fi
        fi
    }

    PROMPT_COMMAND="${PROMPT_COMMAND:-}"
    PROMPT_COMMAND="${PROMPT_COMMAND//rotate_shrimp; /}"
    PROMPT_COMMAND="${PROMPT_COMMAND//rotate_shrimp/}"
    if [[ "$PROMPT_COMMAND" != *update_prompt* ]]; then
        PROMPT_COMMAND="update_prompt${PROMPT_COMMAND:+; $PROMPT_COMMAND}"
    fi

    build_prompt() {
        if [ "$USE_EMOJI" = true ]; then
            # Realign cursor after the emoji
            PS1='${CURRENT_SHRIMP}\[\033[3G\] '
        else
            PS1='${CURRENT_SHRIMP} '
        fi
        PS1+='\[\033[0;90m\][\t]\[\033[00m\] '        # Timestamp [HH:MM:SS] (gray)
        PS1+='\[\033[01;32m\]\u@\h\[\033[00m\]:'      # user@host (green)
        PS1+='\[\033[01;34m\]\w\[\033[00m\]'          # Current directory (blue)
        PS1+='${PROMPT_GIT}'                         # Branch, hash, extras
        PS1+='${PROMPT_BREAK}\$ '                     # Prompt
    }
    build_prompt
fi
