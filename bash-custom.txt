: "${USE_EMOJI:=true}"

# Use a UTF-8 locale when the session has none
if [ "$(locale charmap 2>/dev/null)" != "UTF-8" ]; then
    utf8_locale=$(locale -a 2>/dev/null | grep -ixE 'c\.utf-?8' | head -n 1)
    if [ -z "$utf8_locale" ]; then
        utf8_locale=$(locale -a 2>/dev/null | grep -ixE 'en_us\.utf-?8' | head -n 1)
    fi
    if [ -n "$utf8_locale" ]; then
        export LC_CTYPE="$utf8_locale"
    fi
    unset utf8_locale
fi

if [ -n "$ZSH_VERSION" ]; then
    CURRENT_SHELL="zsh"
    RC_FILE="$HOME/.zshrc"
    eval 'CUSTOM_DIR="${${(%):-%x}:A:h}"'
else
    CURRENT_SHELL="bash"
    RC_FILE="$HOME/.bashrc"
    CUSTOM_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" 2>/dev/null && pwd)"
fi

# ~*x_X- SHRIMP ROTATION -X_x*~
set_shrimps() {
    if [ "$USE_EMOJI" = true ]; then
        SHRIMPS=("🦐" "🍤")
    else
        SHRIMPS=("[>]" "[<]")
    fi
}
set_shrimps
SHRIMP_INDEX=0

rotate_shrimp() {
    # zsh arrays start at 1, bash arrays start at 0
    if [ "$CURRENT_SHELL" = "zsh" ]; then
        CURRENT_SHRIMP="${SHRIMPS[$(( SHRIMP_INDEX + 1 ))]}"
    else
        CURRENT_SHRIMP="${SHRIMPS[$SHRIMP_INDEX]}"
    fi
    SHRIMP_INDEX=$(( (SHRIMP_INDEX + 1) % ${#SHRIMPS[@]} ))
}

# Pop a specific stash
gpopn() {
    if [ -z "$1" ]; then
        echo "Usage: gpopn <stash-number> or gpn <stash-number>"
        echo "Example: gpopn 0"
        echo "Run glist or gsl to see all stashes"
        return 1
    fi
    git stash pop "stash@{$1}"
}
alias gpn='gpopn'

toggle_emoji() {
    if [ "$USE_EMOJI" = true ]; then
        USE_EMOJI=false
    else
        USE_EMOJI=true
    fi
    set_shrimps
    build_prompt

    local begin='# >>> helpful-bash-additions >>>'
    local end='# <<< helpful-bash-additions <<<'
    local state="enabled"
    [ "$USE_EMOJI" = true ] || state="disabled"
    if grep -qxF "$begin" "$RC_FILE" 2>/dev/null; then
        sed -i.tmp "/^$begin\$/,/^$end\$/ s/^USE_EMOJI=.*/USE_EMOJI=$USE_EMOJI/" "$RC_FILE"
        rm -f "${RC_FILE}.tmp"
        echo "Emoji $state"
    else
        echo "Emoji $state for this session only (no setup.sh block in $RC_FILE)"
    fi
}

bupdate() {
    if ! git -C "$CUSTOM_DIR" rev-parse --git-dir > /dev/null 2>&1; then
        echo "Can not check for updates: $CUSTOM_DIR is not a git clone"
        return 1
    fi
    echo "Checking for updates..."
    if ! git -C "$CUSTOM_DIR" fetch --quiet 2> /dev/null; then
        echo "Could not check for updates (git fetch failed). Are you online?"
        return 1
    fi
    local behind answer
    behind=$(git -C "$CUSTOM_DIR" rev-list --count 'HEAD..@{upstream}' 2> /dev/null)
    if [ -z "$behind" ]; then
        echo "Can not check for updates: this clone's branch does not track a remote branch"
        return 1
    fi
    if [ "$behind" -eq 0 ]; then
        echo "Already up to date"
        return 0
    fi
    printf "Update available. Download? (y/n) "
    read -r answer
    case "$answer" in
        y|Y)
            if git -C "$CUSTOM_DIR" pull --ff-only; then
                echo "Updated. Run bsource or open a new terminal to use it."
            else
                echo "Update failed. Nothing was changed. See git's message above."
                return 1
            fi
            ;;
        *)
            echo "Not updated"
            ;;
    esac
}

safe_git() {
    local overrides=() key
    while IFS= read -r key; do
        overrides+=(-c "$key=")
    done < <(git config --name-only --get-regexp '^filter\..*\.(clean|process)$' 2>/dev/null)
    git --no-optional-locks -c core.fsmonitor=false -c core.hooksPath=/dev/null \
        -c log.showSignature=false "${overrides[@]}" "$@"
}

git_state() {
    PROMPT_BRANCH=""      # Branch name, or "detached"
    PROMPT_HASH=""        # Short commit hash, empty before the first commit
    PROMPT_AHEAD=0
    PROMPT_BEHIND=0
    PROMPT_STASH=0
    PROMPT_CONFLICT=0
    PROMPT_STATE=""       # Staged, dirty or clean, empty outside a repo

    local line staged=0 dirty=0 in_repo=0
    while IFS= read -r line; do
        in_repo=1
        case "$line" in
            "# branch.oid (initial)")
                ;;
            "# branch.oid "*)
                PROMPT_HASH="${line#\# branch.oid }"
                PROMPT_HASH="${PROMPT_HASH:0:7}"
                ;;
            "# branch.head "*)
                PROMPT_BRANCH="${line#\# branch.head }"
                ;;
            "# branch.ab "*)
                # Rendered as # branch.ab +2 -1
                line="${line#\# branch.ab +}"
                PROMPT_AHEAD="${line%% *}"
                PROMPT_BEHIND="${line##*-}"
                ;;
            "# stash "*)
                PROMPT_STASH="${line#\# stash }"
                ;;
            "u "*)
                PROMPT_CONFLICT=1
                ;;
            "? "*)
                dirty=1
                ;;
            [12]" "*)
                # Rendered as 1 M. ...  (first letter staged, second unstaged)
                if [ "${line:2:1}" != "." ]; then staged=1; fi
                if [ "${line:3:1}" != "." ]; then dirty=1; fi
                ;;
        esac
    done < <(safe_git status --porcelain=v2 --branch --show-stash \
                 --ignore-submodules=dirty 2>/dev/null)

    if [ "$in_repo" = 0 ]; then
        return
    fi
    if [ "$PROMPT_BRANCH" = "(detached)" ]; then
        PROMPT_BRANCH="detached"
    fi
    if [ "$staged" = 1 ]; then
        PROMPT_STATE="staged"
    elif [ "$dirty" = 1 ]; then
        PROMPT_STATE="dirty"
    else
        PROMPT_STATE="clean"
    fi
}

if [ "$CURRENT_SHELL" = "zsh" ]; then
    autoload -Uz add-zsh-hook
    if [[ "$(whence -f precmd 2>/dev/null)" == *rotate_shrimp* ]]; then
        unfunction precmd
    fi
    typeset -ga psvar

    update_prompt() {
        rotate_shrimp
        git_state
        local arrows=""
        if [ "$PROMPT_AHEAD" -gt 0 ]; then arrows+=" ↑$PROMPT_AHEAD"; fi
        if [ "$PROMPT_BEHIND" -gt 0 ]; then arrows+=" ↓$PROMPT_BEHIND"; fi
        psvar=("$CURRENT_SHRIMP" "" "" "" "$arrows" "" "" "" "")
        if [ -n "$PROMPT_BRANCH" ]; then psvar[2]="($PROMPT_BRANCH)"; fi
        if [ -n "$PROMPT_HASH" ]; then psvar[3]=" $PROMPT_HASH"; fi
        if [ "$PROMPT_CONFLICT" = 1 ]; then psvar[4]="1"; fi
        case "$PROMPT_STATE" in
            staged) psvar[6]="1" ;;
            dirty)  psvar[7]="1" ;;
            clean)  psvar[8]="1" ;;
        esac
        if [ "$PROMPT_STASH" -gt 0 ]; then psvar[9]="stash:$PROMPT_STASH"; fi
    }
    add-zsh-hook precmd update_prompt

    build_prompt() {
        PROMPT='%1v '                                 # Shrimp
        PROMPT+='%F{240}[%D{%H:%M:%S}]%f '            # Timestamp [HH:MM:SS] (gray)
        PROMPT+='%F{green}%B%n@%m%b%f:'               # user@host (green bold)
        PROMPT+='%F{blue}%B%~%b%f'                    # Current directory (blue bold)
        PROMPT+='%(6V.%F{yellow}%B.)'                 # Staged: yellow bold
        PROMPT+='%(7V.%F{red}.)'                      # Unstaged/untracked: red
        PROMPT+='%(8V.%F{green}.)'                    # Clean: green
        PROMPT+=' %2v%3v'                             # (branch) hash
        PROMPT+='%(4V. %F{magenta}%BCONFLICT%b%f.)'   # Merge conflict
        PROMPT+='%5v'                                 # Ahead/behind arrows
        PROMPT+='%(9V. %F{cyan}%9v%f.)'               # stash:N
        PROMPT+='%f%b$ '                              # Reset color + prompt
    }
    build_prompt
else
    update_prompt() {
        rotate_shrimp
        git_state
        local s=$'\001' e=$'\002' esc=$'\033'
        local reset="${s}${esc}[00m${e}"
        PROMPT_GIT=""
        case "$PROMPT_STATE" in
            staged) PROMPT_GIT+="${s}${esc}[1;33m${e}" ;;   # Yellow (staged changes)
            dirty)  PROMPT_GIT+="${s}${esc}[0;31m${e}" ;;   # Red (unstaged/untracked)
            clean)  PROMPT_GIT+="${s}${esc}[0;32m${e}" ;;   # Green (clean)
        esac
        PROMPT_GIT+=" "
        if [ -n "$PROMPT_BRANCH" ]; then PROMPT_GIT+="($PROMPT_BRANCH)"; fi
        if [ -n "$PROMPT_HASH" ]; then PROMPT_GIT+=" $PROMPT_HASH"; fi
        if [ "$PROMPT_CONFLICT" = 1 ]; then
            PROMPT_GIT+=" ${s}${esc}[1;35m${e}CONFLICT${reset}"
        fi
        if [ "$PROMPT_AHEAD" -gt 0 ]; then PROMPT_GIT+=" ↑$PROMPT_AHEAD"; fi
        if [ "$PROMPT_BEHIND" -gt 0 ]; then PROMPT_GIT+=" ↓$PROMPT_BEHIND"; fi
        if [ "$PROMPT_STASH" -gt 0 ]; then
            PROMPT_GIT+=" ${s}${esc}[0;36m${e}stash:$PROMPT_STASH${reset}"
        fi
        PROMPT_GIT+="$reset"
    }

    PROMPT_COMMAND="${PROMPT_COMMAND//rotate_shrimp; /}"
    PROMPT_COMMAND="${PROMPT_COMMAND//rotate_shrimp/}"
    if [[ "$PROMPT_COMMAND" != *update_prompt* ]]; then
        PROMPT_COMMAND="update_prompt${PROMPT_COMMAND:+; $PROMPT_COMMAND}"
    fi

    build_prompt() {
        if [ "$USE_EMOJI" = true ]; then
            # Realign cursor after the emoji
            PS1='${CURRENT_SHRIMP}\[\033[3G\] '
        else
            PS1='${CURRENT_SHRIMP} '
        fi
        PS1+='\[\033[0;90m\][\t]\[\033[00m\] '        # Timestamp [HH:MM:SS] (gray)
        PS1+='\[\033[01;32m\]\u@\h\[\033[00m\]:'      # user@host (green)
        PS1+='\[\033[01;34m\]\w\[\033[00m\]'          # Current directory (blue)
        PS1+='${PROMPT_GIT}'                         # Branch, hash, extras
        PS1+='\$ '                                    # Prompt
    }
    build_prompt
fi
